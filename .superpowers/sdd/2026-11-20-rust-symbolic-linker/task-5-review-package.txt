 use eframe::egui;
+use windows_sys::Win32::Foundation::{HANDLE, CloseHandle};
+use windows_sys::Win32::Security::{GetTokenInformation, TokenElevation, TOKEN_ELEVATION, TOKEN_QUERY};
+use windows_sys::Win32::System::Threading::{GetCurrentProcess, OpenProcessToken};
 
 mod config;
 mod linker;
 mod executor;
 mod app;
 
+fn is_elevated() -> bool {
+    unsafe {
+        let mut handle: HANDLE = 0;
+        if OpenProcessToken(GetCurrentProcess(), TOKEN_QUERY, &mut handle) == 0 {
+            return false;
+        }
+        let mut elevation: TOKEN_ELEVATION = std::mem::zeroed();
+        let mut size = std::mem::size_of::<TOKEN_ELEVATION>() as u32;
+        let result = GetTokenInformation(handle, TokenElevation, &mut elevation as *mut _ as *mut _, size, &mut size);
+        CloseHandle(handle);
+        result != 0 && elevation.TokenIsElevated != 0
+    }
+}
+
+fn get_elevation_error_text() -> String {
+    "Administrator rights are required. PowerShell or the executable must be opened with elevated privileges.".to_string()
+}
+
 fn main() -> Result<(), eframe::Error> {
+    if !is_elevated() {
+        eprintln!("{}", get_elevation_error_text());
+        std::process::exit(1);
+    }
     let options = eframe::NativeOptions {
         viewport: egui::ViewportBuilder::default().with_inner_size([800.0, 600.0]),
         ..Default::default()
     };
     eframe::run_native(
         "Symbolic Link Creator v2.8",
         options,
         Box::new(|_cc| Box::<app::SymbolicLinkApp>::default()),
     )
 }
