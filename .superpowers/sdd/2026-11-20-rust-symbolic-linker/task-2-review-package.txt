+pub fn validate_source(source: &Path, mode: LinkMode) -> Result<(), AppError> {
+    if !source.exists() {
+        return Err(AppError::Validation(format!("Source path does not exist: {}", source.display())));
+    }
+    match mode {
+        LinkMode::File => {
+            if !source.is_file() {
+                return Err(AppError::NotAFile(format!("Source is not a file: {}", source.display())));
+            }
+        }
+        LinkMode::Directory { .. } => {
+            if !source.is_dir() {
+                return Err(AppError::NotADirectory(format!("Source is not a directory: {}", source.display())));
+            }
+        }
+    }
+    Ok(())
+}
+
+pub fn validate_destination(destination: &Path) -> Result<(), AppError> {
+    let parent = destination.parent().ok_or_else(|| {
+        AppError::MissingParent(format!("Destination has no parent directory: {}", destination.display()))
+    })?;
+    if !parent.exists() {
+        return Err(AppError::MissingParent(format!("Destination parent does not exist: {}", parent.display())));
+    }
+    if !parent.is_dir() {
+        return Err(AppError::NotADirectory(format!("Destination parent is not a directory: {}", parent.display())));
+    }
+    Ok(())
+}
+
+pub fn validate_paths(source: &Path, destination: &Path) -> Result<(), AppError> {
+    // Ensure destination is not inside source
+    if destination.starts_with(source) {
+        return Err(AppError::Overlap(format!("Destination {} is inside source {}", destination.display(), source.display())));
+    }
+    // Ensure source is not inside destination (optional but good)
+    if source.starts_with(destination) {
+        return Err(AppError::Overlap(format!("Source {} is inside destination {}", source.display(), destination.display())));
+    }
+    Ok(())
+}

+pub fn plan_links(source: &Path, destination: &Path, mode: LinkMode, overwrite: bool) -> Result<LinkPlan, AppError> {
+    validate_source(source, mode.clone())?;
+    validate_destination(destination)?;
+    validate_paths(source, destination)?;
+
+    let mut actions = Vec::new();
+    match mode {
+        LinkMode::File => {
+            let dest_path = destination.join(source.file_name().unwrap());
+            // Check destination collision
+            if dest_path.exists() {
+                let metadata = std::fs::symlink_metadata(&dest_path)?;
+                if metadata.file_type().is_symlink() {
+                    if overwrite {
+                        // Overwrite symlink: we will replace it later
+                        // For planning, we still create action; execution will delete existing symlink
+                    } else {
+                        // Skip: no action added
+                        return Ok(LinkPlan { actions });
+                    }
+                } else {
+                    // Regular file/directory conflict
+                    return Err(AppError::Conflict(format!("Regular file/directory already exists at {}", dest_path.display())));
+                }
+            }
+            actions.push((PlannedAction::CreateFileLink, source.to_path_buf(), dest_path));
+        }
+        LinkMode::Directory { recursive } => {
+            if recursive {
+                // Walk directories recursively
+                for entry in WalkDir::new(source).follow_links(false) {
+                    let entry = entry.map_err(|e| AppError::Io(std::io::Error::new(std::io::ErrorKind::Other, e)))?;
+                    let rel_path = entry.path().strip_prefix(source).unwrap();
+                    let dest_path = destination.join(rel_path);
+                    if entry.file_type().is_dir() {
+                        // Directory: ensure directory exists at destination
+                        if !dest_path.exists() {
+                            actions.push((PlannedAction::CreateDirectory, source.to_path_buf(), dest_path));
+                        }
+                    } else if entry.file_type().is_file() {
+                        // File: plan file link
+                        if dest_path.exists() {
+                            let metadata = std::fs::symlink_metadata(&dest_path)?;
+                            if metadata.file_type().is_symlink() {
+                                if overwrite {
+                                    // Overwrite symlink
+                                } else {
+                                    // Skip
+                                    continue;
+                                }
+                            } else {
+                                return Err(AppError::Conflict(format!("Regular file/directory already exists at {}", dest_path.display())));
+                            }
+                        }
+                        actions.push((PlannedAction::CreateFileLink, entry.path().to_path_buf(), dest_path));
+                    }
+                }
+            } else {
+                // Top-level mode: only immediate files
+                for entry in std::fs::read_dir(source)? {
+                    let entry = entry?;
+                    if entry.file_type()?.is_file() {
+                        let dest_path = destination.join(entry.file_name());
+                        if dest_path.exists() {
+                            let metadata = std::fs::symlink_metadata(&dest_path)?;
+                            if metadata.file_type().is_symlink() {
+                                if overwrite {
+                                    // Overwrite symlink
+                                } else {
+                                    continue;
+                                }
+                            } else {
+                                return Err(AppError::Conflict(format!("Regular file/directory already exists at {}", dest_path.display())));
+                            }
+                        }
+                        actions.push((PlannedAction::CreateFileLink, entry.path(), dest_path));
+                    }
+                }
+            }
+        }
+    }
+    Ok(LinkPlan { actions })
+}
