+use std::path::PathBuf;
+use crate::linker::{LinkPlan, PlannedAction};
+
+#[derive(Debug, Default)]
+pub struct LinkSummary {
+    pub created_directories: u32,
+    pub created_file_links: u32,
+    pub skipped: u32,
+    pub failures: Vec<(PathBuf, String)>,
+}
+
+pub fn execute_plan(plan: &LinkPlan) -> LinkSummary {
+    let mut summary = LinkSummary::default();
+    
+    for (action, source, dest) in &plan.actions {
+        // Recheck destination immediately before mutation
+        match std::fs::symlink_metadata(dest) {
+            Ok(metadata) => {
+                let file_type = metadata.file_type();
+                if file_type.is_symlink() {
+                    // Destination is a symlink - we can replace it (overwrite was true in planning)
+                    // Remove existing symlink first
+                    if let Err(e) = std::fs::remove_file(dest) {
+                        summary.failures.push((dest.clone(), format!("Failed to remove existing symlink: {}", e)));
+                        continue;
+                    }
+                } else if file_type.is_file() || file_type.is_dir() {
+                    // Regular file or directory - skip (should not happen if planner worked correctly)
+                    summary.skipped += 1;
+                    continue;
+                } else {
+                    // Other type (junction, etc.) - skip and record failure
+                    summary.failures.push((dest.clone(), format!("Destination is a reparse point (not a symlink): {}", dest.display())));
+                    continue;
+                }
+            }
+            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
+                // Destination does not exist - proceed with creation
+            }
+            Err(e) => {
+                // Other error checking destination
+                summary.failures.push((dest.clone(), format!("Failed to check destination: {}", e)));
+                continue;
+            }
+        }
+        
+        // Now perform the action
+        match action {
+            PlannedAction::CreateDirectory => {
+                if let Err(e) = std::fs::create_dir(dest) {
+                    summary.failures.push((dest.clone(), format!("Failed to create directory: {}", e)));
+                } else {
+                    summary.created_directories += 1;
+                }
+            }
+            PlannedAction::CreateFileLink => {
+                // Use Windows symlink_file API
+                match std::os::windows::fs::symlink_file(source, dest) {
+                    Ok(_) => {
+                        summary.created_file_links += 1;
+                    }
+                    Err(e) => {
+                        summary.failures.push((dest.clone(), format!("Failed to create file symlink: {}", e)));
+                    }
+                }
+            }
+            PlannedAction::CreateDirectoryLink => {
+                // Use Windows symlink_dir API
+                match std::os::windows::fs::symlink_dir(source, dest) {
+                    Ok(_) => {
+                        // Note: we don't have a separate count for directory links in summary
+                        // According to spec, LinkSummary has created_directory and created_file_link counts
+                        // Directory links might be counted as created_directories? Or we need to add field.
+                        // For now, treat as directory creation count.
+                        summary.created_directories += 1;
+                    }
+                    Err(e) => {
+                        summary.failures.push((dest.clone(), format!("Failed to create directory symlink: {}", e)));
+                    }
+                }
+            }
+        }
+    }
+    
+    summary
+}

+    #[test]
+    fn execute_plan_creates_file_symlink() {
+        let tmp = tempdir().unwrap();
+        let source = tmp.path().join("source.txt");
+        fs::write(&source, "content").unwrap();
+        let dest_dir = tmp.path().join("dest");
+        fs::create_dir(&dest_dir).unwrap();
+        let dest_path = dest_dir.join("source.txt");
+        let plan = LinkPlan {
+            actions: vec![(PlannedAction::CreateFileLink, source.clone(), dest_path.clone())],
+        };
+        let summary = execute_plan(&plan);
+        // Expect one created file link if privilege allows, otherwise failures
+        if summary.failures.is_empty() {
+            assert_eq!(summary.created_file_links, 1);
+            assert_eq!(summary.created_directories, 0);
+            assert_eq!(summary.skipped, 0);
+            // Verify symlink exists
+            match fs::read_link(&dest_path) {
+                Ok(target) => assert_eq!(target, source),
+                Err(e) if e.raw_os_error() == Some(1314) => {
+                    // Privilege not held; but summary should have failures
+                    // Actually execute_plan would have recorded permission error in failures
+                    // So this branch shouldn't happen
+                }
+                Err(e) => panic!("Failed to read symlink: {}", e),
+            }
+        } else {
+            // Permission error expected
+            assert!(summary.failures.len() > 0);
+            assert!(summary.failures[0].1.contains("Failed to create file symlink"));
+        }
+    }
